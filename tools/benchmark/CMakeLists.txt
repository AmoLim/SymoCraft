cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(SymoCraftBenchmark VERSION 0.1.0 LANGUAGES C CXX)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The benchmark runner requires Windows x64 and MSVC.")
    endif()
    if(DEFINED MSVC_CXX_ARCHITECTURE_ID AND NOT MSVC_CXX_ARCHITECTURE_ID STREQUAL "x64")
        message(FATAL_ERROR "The benchmark runner requires x64, not ${MSVC_CXX_ARCHITECTURE_ID}.")
    endif()
    include(CTest)
    get_filename_component(SYMOCRAFT_REPOSITORY_ROOT "../.." ABSOLUTE)
    include("${SYMOCRAFT_REPOSITORY_ROOT}/cmake/ValidateMsvcEnvironment.cmake")
    include("${SYMOCRAFT_REPOSITORY_ROOT}/cmake/ThirdPartyYaml.cmake")
    if(CMAKE_BUILD_TYPE STREQUAL "Release")
        set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION ".")
        include(InstallRequiredSystemLibraries)
    endif()
endif()

add_library(benchmark_core STATIC src/platform.cpp src/runner.cpp)
target_include_directories(benchmark_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(benchmark_core PUBLIC cxx_std_20)
target_compile_definitions(benchmark_core PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(benchmark_core PRIVATE /W4 /utf-8 /permissive-)
set_target_properties(benchmark_core PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(benchmark_core PUBLIC symocraft_yaml PRIVATE bcrypt advapi32 powrprof user32)

add_executable(SymoCraftBenchmark WIN32 src/main.cpp src/gui.cpp runner.manifest)
target_compile_options(SymoCraftBenchmark PRIVATE /W4 /utf-8 /permissive-)
target_link_libraries(SymoCraftBenchmark PRIVATE benchmark_core comctl32 comdlg32 shell32 ole32)
set_target_properties(SymoCraftBenchmark PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin" CXX_EXTENSIONS OFF)
install(TARGETS SymoCraftBenchmark RUNTIME DESTINATION .)
install(FILES "${SYMOCRAFT_REPOSITORY_ROOT}/docs/project/benchmark/使用与交付.md" DESTINATION . RENAME "BENCHMARK-README.zh-CN.md")
install(FILES "${SYMOCRAFT_REPOSITORY_ROOT}/docs/third-party-inventory.md" DESTINATION . RENAME "THIRD-PARTY-INVENTORY.md")
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
