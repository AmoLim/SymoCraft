cmake_minimum_required(VERSION 3.22)
project(SymoCraftQ06Presample LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "The Q06 helper requires Windows x64 MSVC Release and actual Release imports")
endif()
get_filename_component(SYMOCRAFT_REPOSITORY_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(SYMOCRAFT_Q06_PRODUCTION_BUILD "" CACHE PATH "Actual Release build containing the existing runner core")
set(SYMOCRAFT_Q06_BENCHMARK_CORE_LIBRARY "" CACHE FILEPATH "Exact existing Release benchmark_core.lib")
set(SYMOCRAFT_Q06_YAML_LIBRARY "" CACHE FILEPATH "Exact existing Release yaml-cpp archive")
if(NOT EXISTS "${SYMOCRAFT_Q06_PRODUCTION_BUILD}/CMakeCache.txt")
    message(FATAL_ERROR "Declare the actual Release production build; this helper never rebuilds its core")
endif()
file(STRINGS "${SYMOCRAFT_Q06_PRODUCTION_BUILD}/CMakeCache.txt" production_cache)
foreach(entry IN LISTS production_cache)
    if(entry MATCHES "^CMAKE_BUILD_TYPE:STRING=(.*)$")
        set(production_configuration "${CMAKE_MATCH_1}")
    elseif(entry MATCHES "^CMAKE_HOME_DIRECTORY:INTERNAL=(.*)$")
        get_filename_component(production_source "${CMAKE_MATCH_1}" REALPATH)
    endif()
endforeach()
get_filename_component(repository_source "${SYMOCRAFT_REPOSITORY_ROOT}" REALPATH)
if(NOT production_configuration STREQUAL "Release" OR NOT production_source STREQUAL repository_source)
    message(FATAL_ERROR "The imported build must be this repository's Release build")
endif()
foreach(input IN ITEMS SYMOCRAFT_Q06_BENCHMARK_CORE_LIBRARY SYMOCRAFT_Q06_YAML_LIBRARY)
    if(NOT IS_ABSOLUTE "${${input}}" OR NOT EXISTS "${${input}}")
        message(FATAL_ERROR "Declare an exact existing absolute library path: ${input}")
    endif()
endforeach()
add_library(q06_existing_core STATIC IMPORTED)
set_target_properties(q06_existing_core PROPERTIES IMPORTED_LOCATION "${SYMOCRAFT_Q06_BENCHMARK_CORE_LIBRARY}")
add_library(q06_existing_yaml STATIC IMPORTED)
set_target_properties(q06_existing_yaml PROPERTIES IMPORTED_LOCATION "${SYMOCRAFT_Q06_YAML_LIBRARY}")
include("${SYMOCRAFT_REPOSITORY_ROOT}/cmake/ThirdPartyHeaders.cmake")
symocraft_header_view(q06_yaml_headers yaml-cpp)
add_executable(SymoCraftQ06VerifyCapture verify-capture.cpp "${SYMOCRAFT_REPOSITORY_ROOT}/cmake/windows-utf8.manifest")
target_compile_features(SymoCraftQ06VerifyCapture PRIVATE cxx_std_20)
target_compile_options(SymoCraftQ06VerifyCapture PRIVATE /W4 /utf-8 /permissive-)
target_compile_definitions(SymoCraftQ06VerifyCapture PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(SymoCraftQ06VerifyCapture PRIVATE "${SYMOCRAFT_REPOSITORY_ROOT}/tools/benchmark/include")
target_link_libraries(SymoCraftQ06VerifyCapture PRIVATE q06_existing_core q06_existing_yaml q06_yaml_headers bcrypt advapi32 powrprof user32)
set_target_properties(SymoCraftQ06VerifyCapture PROPERTIES CXX_EXTENSIONS OFF MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
file(SHA256 "${SYMOCRAFT_Q06_BENCHMARK_CORE_LIBRARY}" core_sha)
file(SHA256 "${SYMOCRAFT_Q06_YAML_LIBRARY}" yaml_sha)
file(SHA256 "${SYMOCRAFT_Q06_PRODUCTION_BUILD}/CMakeCache.txt" cache_sha)
file(TO_CMAKE_PATH "${SYMOCRAFT_Q06_BENCHMARK_CORE_LIBRARY}" core_path)
file(TO_CMAKE_PATH "${SYMOCRAFT_Q06_YAML_LIBRARY}" yaml_path)
file(WRITE "${CMAKE_BINARY_DIR}/production-imports.json"
    "{\n  \"configuration\": \"Release\",\n  \"benchmark_core\": {\"path\": \"${core_path}\", \"sha256\": \"${core_sha}\"},\n  \"yaml\": {\"path\": \"${yaml_path}\", \"sha256\": \"${yaml_sha}\"},\n  \"production_cache_sha256\": \"${cache_sha}\"\n}\n")
